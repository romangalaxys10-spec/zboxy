    38→// File icon helper
    39→function FileIcon({ file, size = 20 }: { file: ZFile; size?: number }) {
    40→  if (file.type === 'folder') return <Folder className="text-amber-500" style={{ width: size, height: size }} />;
    41→  const cat = getFileCategory(file.mimeType || '', file.name);
    42→  const iconMap: Record<FileCategory, React.ReactNode> = {
    43→    document: <FileText className="text-blue-600" style={{ width: size, height: size }} />,
    44→    spreadsheet: <Table2 className="text-emerald-600" style={{ width: size, height: size }} />,
    45→    presentation: <Presentation className="text-orange-600" style={{ width: size, height: size }} />,
    46→    image: <ImageIcon className="text-purple-500" style={{ width: size, height: size }} />,
    47→    video: <Video className="text-red-500" style={{ width: size, height: size }} />,
    48→    audio: <Music className="text-pink-500" style={{ width: size, height: size }} />,
    49→    pdf: <FileText className="text-red-600" style={{ width: size, height: size }} />,
    50→    archive: <FileArchive className="text-amber-700" style={{ width: size, height: size }} />,
    51→    code: <FileCode className="text-slate-600" style={{ width: size, height: size }} />,
    52→    other: <File className="text-slate-400" style={{ width: size, height: size }} />,
    53→  };
    54→  return <>{iconMap[cat] || iconMap.other}</>;
    55→}

    57→// Thumbnails for office files
    58→function FileThumbnail({ file }: { file: ZFile }) {
    59→  const token = useZboxyStore(s => s.token);
    60→  const ext = file.name.split('.').pop()?.toLowerCase() || '';
    61→  if (file.type === 'folder') {
    62→    return (
    63→      <div className="w-full h-full bg-amber-50 rounded-t-lg flex items-center justify-center">
    64→        <Folder className="w-16 h-16 text-amber-400" />
    65→      </div>
    66→    );
    67→  }
    68→  const cat = getFileCategory(file.mimeType || '', file.name);
    69→  if (cat === 'image') {
    70→    return <img src={`/api/zboxy/files/download?id=${file.id}&token=${token}`} alt={file.name} className="w-full h-full object-cover rounded-t-lg" />;
    71→  }
    72→  const bgColors: Record<string, string> = {
    73→    document: 'bg-blue-50', spreadsheet: 'bg-emerald-50', presentation: 'bg-orange-50',
    74→    pdf: 'bg-red-50', code: 'bg-slate-50', video: 'bg-red-50', audio: 'bg-pink-50',
    75→  };
    76→  const labels: Record<string, string> = {
    77→    zdoc: 'DOC', zsheet: 'SHEET', zslide: 'SLIDE',
    78→    pdf: 'PDF', doc: 'DOC', docx: 'DOCX', xls: 'XLS', xlsx: 'XLSX',
    79→    ppt: 'PPT', pptx: 'PPTX', csv: 'CSV',
    80→  };
    81→  return (
    82→    <div className={`w-full h-2/3 ${bgColors[cat] || 'bg-slate-50'} rounded-t-lg flex items-center justify-center relative`}>
    83→      <FileIcon file={file} size={48} />
    84→      {labels[ext] && (
    85→        <span className="absolute bottom-2 right-2 text-[10px] font-bold text-slate-500 bg-white/80 px-1.5 py-0.5 rounded">
    86→          {labels[ext]}
    87→        </span>
    88→      )}
    89→    </div>
    90→  );
    91→}

   207→    const total = Array.from(fileList).length;
   208→    const createdFolders = new Set<string>();

   211→    for (const file of Array.from(fileList)) {
   212→      const relativePath = (file as File & { webkitRelativePath?: string }).webkitRelativePath || '';

   214→      if (parts.length > 1) {
   215→        // Build up folder paths and create them
   216→        let folderPath = '';
   217→        for (let i = 0; i < parts.length - 1; i++) {
   218→          folderPath = folderPath === '' ? '/' + parts[i] : folderPath + '/' + parts[i];
   219→          if (!createdFolders.has(folderPath)) {
   220→            createdFolders.add(folderPath);
   221→            try {
   222→              await fetch('/api/zboxy/folders', {
   223→                method: 'POST',
   224→                headers: { 'Content-Type': 'application/json', 'x-zboxy-token': token },
   225→                body: JSON.stringify({ name: parts[i], parent: folderPath.substring(0, folderPath.lastIndexOf('/')) || '/' }),
   226→              });
   227→            } catch { /* folder may already exist */ }
   228→          }
   229→        }
   230→      }
   231→    }
   232→
   233→    // Second pass: upload files to correct folders
   234→    for (const file of Array.from(fileList)) {
   235→      const relativePath = (file as File & { webkitRelativePath?: string }).webkitRelativePath || '';

   237→      const fileName = parts[parts.length - 1];
   238→      const folderPath = parts.length > 1
   239→        ? '/' + parts.slice(0, parts.length - 1).join('/')
   240→        : currentPath;

   242→      const formData = new FormData();
   243→      formData.append('file', file);
   244→      formData.append('folder', folderPath);
   245→      try {
   246→        await fetch('/api/zboxy/files', { method: 'POST', headers: { 'x-zboxy-token': token }, body: formData });
   247→      } catch { toast.error(`${fileName}: Upload failed`); }
   248→      done++;
   249→      setUploadProgress(Math.round((done / total) * 100));
   250→    }
   251→    setUploading(false);
   252→    setUploadProgress(0);
   253→    fetchFiles();
