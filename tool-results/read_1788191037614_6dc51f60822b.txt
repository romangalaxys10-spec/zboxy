    28→// File icon helper
    29→function FileIcon({ file, size = 20 }: { file: ZFile; size?: number }) {
    30→  if (file.type === 'folder') return <Folder className="text-amber-500" style={{ width: size, height: size }} />;
    31→  const cat = getFileCategory(file.mimeType || '', file.name);
    32→  const iconMap: Record<FileCategory, React.ReactNode> = {
    33→    document: <FileText className="text-blue-600" style={{ width: size, height: size }} />,
    34→    spreadsheet: <Table2 className="text-emerald-600" style={{ width: size, height: size }} />,
    35→    presentation: <Presentation className="text-orange-600" style={{ width: size, height: size }} />,
    36→    image: <ImageIcon className="text-purple-500" style={{ width: size, height: size }} />,
    37→    video: <Video className="text-red-500" style={{ width: size, height: size }} />,
    38→    audio: <Music className="text-pink-500" style={{ width: size, height: size }} />,
    39→    pdf: <FileText className="text-red-600" style={{ width: size, height: size }} />,
    40→    archive: <FileArchive className="text-amber-700" style={{ width: size, height: size }} />,
    41→    code: <FileCode className="text-slate-600" style={{ width: size, height: size }} />,
    42→    other: <File className="text-slate-400" style={{ width: size, height: size }} />,
    43→  };
    44→  return <>{iconMap[cat] || iconMap.other}</>;
    45→}

    47→// Thumbnails for office files
    48→function FileThumbnail({ file }: { file: ZFile }) {
    49→  const ext = file.name.split('.').pop()?.toLowerCase() || '';
    50→  if (file.type === 'folder') {
    51→    return (
    52→      <div className="w-full h-full bg-amber-50 rounded-t-lg flex items-center justify-center">
    53→        <Folder className="w-16 h-16 text-amber-400" />
    54→      </div>
    55→    );
    56→  }
    57→  const cat = getFileCategory(file.mimeType || '', file.name);
    58→  if (cat === 'image') {
    59→    return <img src={`/api/zboxy/files/download?id=${file.id}&token=${useZboxyStore.getState().token}`} alt={file.name} className="w-full h-full object-cover rounded-t-lg" />;
    60→  }
    61→  const bgColors: Record<string, string> = {
    62→    document: 'bg-blue-50', spreadsheet: 'bg-emerald-50', presentation: 'bg-orange-50',
    63→    pdf: 'bg-red-50', code: 'bg-slate-50', video: 'bg-red-50', audio: 'bg-pink-50',
    64→  };
    65→  const labels: Record<string, string> = {
    66→    zdoc: 'DOC', zsheet: 'SHEET', zslide: 'SLIDE',
    67→    pdf: 'PDF', doc: 'DOC', docx: 'DOCX', xls: 'XLS', xlsx: 'XLSX',
    68→    ppt: 'PPT', pptx: 'PPTX', csv: 'CSV',
    69→  };
    70→  return (
    71→    <div className={`w-full h-2/3 ${bgColors[cat] || 'bg-slate-50'} rounded-t-lg flex items-center justify-center relative`}>
    72→      <FileIcon file={file} size={48} />
    73→      {labels[ext] && (
    74→        <span className="absolute bottom-2 right-2 text-[10px] font-bold text-slate-500 bg-white/80 px-1.5 py-0.5 rounded">
    75→          {labels[ext]}
    76→        </span>
    77→      )}
    78→    </div>
    79→  );
    80→}

    86→function formatTime(d: string) {

   471→                }} className="gap-1"><Star className="w-3 h-3" /> Star</Button>
   472→                {activeView !== 'trash' && (
   473→                  <Button variant="outline" size="sm" onClick={handleDeleteSelected} className="gap-1 text-red-600"><Trash2 className="w-3 h-3" /> Delete</Button>

   476→              </>
   477→            )}
   478→            <Button variant="outline" size="sm" onClick={() => fileInputRef.current?.click()} className="gap-1.5" disabled={activeView === 'trash'}>
   479→              <Upload className="w-3.5 h-3.5" /> Upload
   480→            </Button>
   481→            <input ref={fileInputRef} type="file" multiple className="hidden" onChange={(e) => e.target.files && handleUpload(e.target.files)} />

   483→              <Button variant="outline" size="sm" onClick={handleEmptyTrash} className="gap-1 text-red-600">Empty Trash</Button>

   485→          </div>
   486→
   487→          {/* Upload progress */}
   488→          {uploading && (
   489→            <div className="px-4 py-2 bg-emerald-50 border-b flex items-center gap-3">
   490→              <div className="flex-1 h-1.5 bg-emerald-200 rounded-full overflow-hidden">
   491→                <div className="h-full bg-emerald-500 rounded-full transition-all duration-300" style={{ width: `${uploadProgress}%` }} />
   492→              </div>
   493→              <span className="text-xs text-emerald-700">{uploadProgress}%</span>
   494→            </div>
   495→          )}
   496→
   497→          {/* Drag overlay */}
   498→          {dragOver && (
   499→            <div className="absolute inset-0 bg-emerald-500/10 border-2 border-dashed border-emerald-500 z-30 flex items-center justify-center pointer-events-none">
   500→              <div className="bg-white rounded-xl p-8 shadow-xl text-center">
   501→                <Upload className="w-12 h-12 text-emerald-500 mx-auto mb-2" />
   502→                <p className="text-lg font-medium text-slate-700">Drop files to upload</p>
   503→              </div>
   504→            </div>
   505→          )}
   506→
   507→          {/* File content */}
   508→          <div className="flex-1 overflow-auto" onClick={() => clearSelection()}>
   509→            {loading && files.length === 0 ? (
   510→              <div className="flex items-center justify-center h-64">
   511→                <div className="animate-spin w-8 h-8 border-2 border-emerald-500 border-t-transparent rounded-full" />
   512→              </div>
   513→            ) : files.length === 0 ? (
   514→              <div className="flex flex-col items-center justify-center h-64 text-center px-4">
   515→                {searchQuery ? (
   516→                  <><Search className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">No results for &quot;{searchQuery}&quot;</p></>
   517→                ) : activeView === 'trash' ? (
   518→                  <><Trash2 className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">Trash is empty</p></>
   519→                ) : activeView === 'starred' ? (
   520→                  <><Star className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">No starred files</p></>
   521→                ) : (
   522→                  <><FolderOpen className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">This folder is empty</p>
   523→                  <p className="text-sm text-slate-400 mt-1">Upload files or create new documents</p></>
   524→                )}
   525→              </div>
   526→            ) : viewMode === 'grid' ? (
   527→              <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 2xl:grid-cols-8 gap-1 p-4">
   528→                {files.map(file => (
   529→                  <ContextMenu key={file.id}>
   530→                    <ContextMenuTrigger>
   531→                      <div
   532→                        className={`group relative border rounded-lg hover:shadow-md transition-all cursor-pointer ${selectedFiles.includes(file.id) ? 'border-emerald-500 bg-emerald-50/50' : 'border-transparent hover:border-slate-200'}`}

   539→                            checked={selectedFiles.includes(file.id)}
   540→                            onCheckedChange={() => toggleSelect(file.id)}
   541→                            className="mt-0.5 shrink-0 opacity-0 group-hover:opacity-100 transition-opacity"

   543→                          />

   549→                        </div>
   550→                      </div>
   551→                    </ContextMenuTrigger>
   552→                    <ContextMenuContent>
   553→                      <ContextMenuItem onClick={() => handleFileDoubleClick(file)}><Eye className="w-4 h-4 mr-2" /> Open</ContextMenuItem>

   556→                      <ContextMenuItem onClick={() => handleDownloadFile(file)}><Download className="w-4 h-4 mr-2" /> Download</ContextMenuItem>
   557→                      <ContextMenuSeparator />
   558→                      <ContextMenuItem className="text-red-600" onClick={() => {
   559→                        if (activeView === 'trash') handleDeletePermanent([file.id]);
   560→                        else { setSelectedFiles([file.id]); handleDeleteSelected(); }
   561→                      }}><Trash2 className="w-4 h-4 mr-2" /> {activeView === 'trash' ? 'Delete permanently' : 'Move to trash'}</ContextMenuItem>
   562→                    </ContextMenuContent>
   563→                  </ContextMenu>
   564→                ))}

   567→              <table className="w-full text-sm">
   568→                <thead className="sticky top-0 bg-white border-b">

   572→                    <th className="text-left py-2 px-3 font-medium text-slate-500 hidden md:table-cell">Owner</th>

   581→                      <ContextMenuTrigger>
   582→                        <tr
   583→                          className={`border-b hover:bg-slate-50 cursor-pointer transition-colors ${selectedFiles.includes(file.id) ? 'bg-emerald-50' : ''}`}

   588→                          <td className="py-1 px-3">
   589→                            <div className="flex items-center gap-3">
   590→                              <FileIcon file={file} />
   591→                              <span className="truncate font-medium">{file.name}</span>
   592→                            </div>
   593→                          </td>
   594→                          <td className="py-1 px-3 text-slate-500 hidden md:table-cell">{user?.name || 'me'}</td>
   595→                          <td className="py-1 px-3 text-slate-500 hidden sm:table-cell">{formatTime(file.updatedAt)}</td>
   596→                          <td className="py-1 px-3 text-slate-500 hidden lg:table-cell">{file.type === 'folder' ? '--' : formatFileSize(file.size)}</td>
   597→                          <td className="px-2">
   598→                            <DropdownMenu>
   599→                              <DropdownMenuTrigger asChild onClick={(e) => e.stopPropagation()}>
   600→                                <Button variant="ghost" size="icon" className="h-8 w-8"><MoreVertical className="w-4 h-4" /></Button>
   601→                              </DropdownMenuTrigger>
   602→                              <DropdownMenuContent align="end">
   603→                                <DropdownMenuItem onClick={() => handleFileDoubleClick(file)}><Eye className="w-4 h-4 mr-2" /> Open</DropdownMenuItem>

   606→                                <DropdownMenuItem onClick={() => handleDownloadFile(file)}><Download className="w-4 h-4 mr-2" /> Download</DropdownMenuItem>
   607→                                <DropdownMenuSeparator />
   608→                                <DropdownMenuItem className="text-red-600" onClick={() => {
   609→                                  if (activeView === 'trash') handleDeletePermanent([file.id]);
   610→                                  else { setSelectedFiles([file.id]); handleDeleteSelected(); }
   611→                                }}><Trash2 className="w-4 h-4 mr-2" /> {activeView === 'trash' ? 'Delete forever' : 'Move to trash'}</DropdownMenuItem>
   612→                              </DropdownMenuContent>
   613→                            </DropdownMenu>
   614→                          </td>
   615→                        </tr>
   616→                      </ContextMenuTrigger>
   617→                      <ContextMenuContent>
   618→                        <ContextMenuItem onClick={() => handleFileDoubleClick(file)}><Eye className="w-4 h-4 mr-2" /> Open</ContextMenuItem>
   619→                        <ContextMenuItem onClick={() => { setRenameDialog({ open: true, fileId: file.id, name: file.name }); setRenameValue(file.name); }}><Edit3 className="w-4 h-4 mr-2" /> Rename</ContextMenuItem>
   620→                        <ContextMenuItem onClick={() => handleDownloadFile(file)}><Download className="w-4 h-4 mr-2" /> Download</ContextMenuItem>
   621→                        <ContextMenuItem className="text-red-600" onClick={() => {
   622→                          if (activeView === 'trash') handleDeletePermanent([file.id]);
   623→                          else { setSelectedFiles([file.id]); handleDeleteSelected(); }
   624→                        }}><Trash2 className="w-4 h-4 mr-2" /> {activeView === 'trash' ? 'Delete forever' : 'Move to trash'}</ContextMenuItem>

   631→          </div>

   634→          <div className="flex items-center justify-between px-4 py-1.5 border-t bg-slate-50 text-xs text-slate-500 shrink-0">

   638→        </main>
   639→
   640→        {/* Detail Panel */}
   641→        {detailPanelOpen && detailFile && (

   644→              <h3 className="font-semibold text-sm">Details</h3>

   647→            <div className="flex flex-col items-center py-4">
   648→              <FileIcon file={detailFile} size={64} />
   649→              <p className="font-medium mt-3 text-center break-all">{detailFile.name}</p>

   677→          <div className="space-y-4 py-2">
   678→            <div className="space-y-2">
   679→              <label className="text-sm font-medium">File Name</label>
   680→              <Input placeholder="Untitled" value={newFileName} onChange={(e) => setNewFileName(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleCreateFile()} />

   682→            <div className="space-y-2">
   683→              <label className="text-sm font-medium">Type</label>
   684→              <div className="grid grid-cols-3 gap-3">
   685→                {[{ type: 'doc' as const, icon: FileText, label: 'Document', desc: 'Rich text editor', color: 'text-blue-600 bg-blue-50' },
   686→                  { type: 'sheet' as const, icon: Table2, label: 'Spreadsheet', desc: 'Cells & formulas', color: 'text-emerald-600 bg-emerald-50' },
   687→                  { type: 'slide' as const, icon: Presentation, label: 'Presentation', desc: 'Slides & themes', color: 'text-orange-600 bg-orange-50' },

   689→                  <button key={opt.type} onClick={() => setNewFileType(opt.type)} className={`p-3 rounded-lg border-2 text-center transition-all ${newFileType === opt.type ? 'border-emerald-500 bg-emerald-50' : 'border-slate-200 hover:border-slate-300'}`}>
   690→                    <opt.icon className={`w-8 h-8 mx-auto mb-1 ${opt.color.split(' ')[0]}`} />

   692→                  </button>

   695→            </div>
   696→          </div>
   697→          <DialogFooter>
   698→            <Button variant="outline" onClick={() => setNewFileDialog(false)}>Cancel</Button>
   699→            <Button onClick={handleCreateFile} disabled={!newFileName.trim()} className="bg-emerald-600 hover:bg-emerald-700">Create</Button>
   700→          </DialogFooter>
   701→        </DialogContent>
   702→      </Dialog>
   703→
   704→      {/* Rename Dialog */}
   705→      <Dialog open={!!renameDialog?.open} onOpenChange={() => setRenameDialog(null)}>

   707→          <DialogHeader><DialogTitle>Rename</DialogTitle></DialogHeader>
   708→          <Input value={renameValue} onChange={(e) => setRenameValue(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleRename()} />
   709→          <DialogFooter>
   710→            <Button variant="outline" onClick={() => setRenameDialog(null)}>Cancel</Button>
   711→            <Button onClick={handleRename} className="bg-emerald-600 hover:bg-emerald-700">Rename</Button>
   712→          </DialogFooter>
   713→        </DialogContent>
   714→      </Dialog>
   715→    </div>
   716→  );
   717→}
