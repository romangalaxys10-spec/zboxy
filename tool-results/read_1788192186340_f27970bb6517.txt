    29→// File icon helper
    30→function FileIcon({ file, size = 20 }: { file: ZFile; size?: number }) {
    31→  if (file.type === 'folder') return <Folder className="text-amber-500" style={{ width: size, height: size }} />;
    32→  const cat = getFileCategory(file.mimeType || '', file.name);
    33→  const iconMap: Record<FileCategory, React.ReactNode> = {
    34→    document: <FileText className="text-blue-600" style={{ width: size, height: size }} />,
    35→    spreadsheet: <Table2 className="text-emerald-600" style={{ width: size, height: size }} />,
    36→    presentation: <Presentation className="text-orange-600" style={{ width: size, height: size }} />,
    37→    image: <ImageIcon className="text-purple-500" style={{ width: size, height: size }} />,
    38→    video: <Video className="text-red-500" style={{ width: size, height: size }} />,
    39→    audio: <Music className="text-pink-500" style={{ width: size, height: size }} />,
    40→    pdf: <FileText className="text-red-600" style={{ width: size, height: size }} />,
    41→    archive: <FileArchive className="text-amber-700" style={{ width: size, height: size }} />,
    42→    code: <FileCode className="text-slate-600" style={{ width: size, height: size }} />,
    43→    other: <File className="text-slate-400" style={{ width: size, height: size }} />,
    44→  };
    45→  return <>{iconMap[cat] || iconMap.other}</>;
    46→}

    48→// Thumbnails for office files
    49→function FileThumbnail({ file }: { file: ZFile }) {
    50→  const ext = file.name.split('.').pop()?.toLowerCase() || '';
    51→  if (file.type === 'folder') {
    52→    return (
    53→      <div className="w-full h-full bg-amber-50 rounded-t-lg flex items-center justify-center">
    54→        <Folder className="w-16 h-16 text-amber-400" />
    55→      </div>
    56→    );
    57→  }
    58→  const cat = getFileCategory(file.mimeType || '', file.name);
    59→  if (cat === 'image') {
    60→    return <img src={`/api/zboxy/files/download?id=${file.id}&token=${useZboxyStore.getState().token}`} alt={file.name} className="w-full h-full object-cover rounded-t-lg" />;
    61→  }
    62→  const bgColors: Record<string, string> = {
    63→    document: 'bg-blue-50', spreadsheet: 'bg-emerald-50', presentation: 'bg-orange-50',
    64→    pdf: 'bg-red-50', code: 'bg-slate-50', video: 'bg-red-50', audio: 'bg-pink-50',
    65→  };
    66→  const labels: Record<string, string> = {
    67→    zdoc: 'DOC', zsheet: 'SHEET', zslide: 'SLIDE',
    68→    pdf: 'PDF', doc: 'DOC', docx: 'DOCX', xls: 'XLS', xlsx: 'XLSX',
    69→    ppt: 'PPT', pptx: 'PPTX', csv: 'CSV',
    70→  };
    71→  return (
    72→    <div className={`w-full h-2/3 ${bgColors[cat] || 'bg-slate-50'} rounded-t-lg flex items-center justify-center relative`}>
    73→      <FileIcon file={file} size={48} />
    74→      {labels[ext] && (
    75→        <span className="absolute bottom-2 right-2 text-[10px] font-bold text-slate-500 bg-white/80 px-1.5 py-0.5 rounded">
    76→          {labels[ext]}
    77→        </span>
    78→      )}
    79→    </div>
    80→  );
    81→}

    83→// Format date
    84→function formatDate(d: string) {
    85→  return new Date(d).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' });
    86→}
    87→function formatTime(d: string) {
    88→  return new Date(d).toLocaleString('en-US', { month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' });
    89→}

   713→          <DialogFooter>
