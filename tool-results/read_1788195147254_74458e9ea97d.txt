    42→    presentation: <Presentation className="text-orange-600" style={{ width: size, height: size }} />,
    43→    image: <ImageIcon className="text-purple-500" style={{ width: size, height: size }} />,
    44→    video: <Video className="text-red-500" style={{ width: size, height: size }} />,
    45→    audio: <Music className="text-pink-500" style={{ width: size, height: size }} />,
    46→    pdf: <FileText className="text-red-600" style={{ width: size, height: size }} />,
    47→    archive: <FileArchive className="text-amber-700" style={{ width: size, height: size }} />,
    48→    code: <FileCode className="text-slate-600" style={{ width: size, height: size }} />,

   471→                <span className="text-xs text-slate-500">{selectedFiles.length} selected</span>
   472→                <Button variant="outline" size="sm" onClick={() => {
   473→                  selectedFiles.forEach(id => handleStar(id));
   474→                }} className="gap-1"><Star className="w-3 h-3" /> Star</Button>
   475→                {activeView !== 'trash' && (
   476→                  <Button variant="outline" size="sm" onClick={handleDeleteSelected} className="gap-1 text-red-600"><Trash2 className="w-3 h-3" /> Delete</Button>
   477→                )}
   478→                <Button variant="ghost" size="sm" onClick={clearSelection}><X className="w-3 h-3" /></Button>
   479→              </>
   480→            )}
   481→            <Button variant="outline" size="sm" onClick={() => fileInputRef.current?.click()} className="gap-1.5" disabled={activeView === 'trash'}>
   482→              <Upload className="w-3.5 h-3.5" /> Upload
   483→            </Button>
   484→            <input ref={fileInputRef} type="file" multiple className="hidden" onChange={(e) => e.target.files && handleUpload(e.target.files)} />
   485→            {activeView === 'trash' && files.length > 0 && (
   486→              <Button variant="outline" size="sm" onClick={handleEmptyTrash} className="gap-1 text-red-600">Empty Trash</Button>
   487→            )}
   488→          </div>
   489→
   490→          {/* Upload progress */}
   491→          {uploading && (
   492→            <div className="px-4 py-2 bg-emerald-50 border-b flex items-center gap-3">
   493→              <div className="flex-1 h-1.5 bg-emerald-200 rounded-full overflow-hidden">
   494→                <div className="h-full bg-emerald-500 rounded-full transition-all duration-300" style={{ width: `${uploadProgress}%` }} />
   495→              </div>
   496→              <span className="text-xs text-emerald-700">{uploadProgress}%</span>
   497→            </div>
   498→          )}
   499→
   500→          {/* Drag overlay */}
   501→          {dragOver && (
   502→            <div className="fixed inset-0 bg-emerald-500/10 border-2 border-dashed border-emerald-500 z-30 flex items-center justify-center pointer-events-none">
   503→              <div className="bg-white rounded-xl p-8 shadow-xl text-center">
   504→                <Upload className="w-12 h-12 text-emerald-500 mx-auto mb-2" />
   505→                <p className="text-lg font-medium text-slate-700">Drop files to upload</p>
   506→              </div>
   507→            </div>
   508→          )}
   509→
   510→          {/* File content */}
   511→          <div className="flex-1 overflow-auto" onClick={() => clearSelection()}>
   512→            {loading && files.length === 0 ? (
   513→              <div className="flex items-center justify-center h-64">
   514→                <div className="animate-spin w-8 h-8 border-2 border-emerald-500 border-t-transparent rounded-full" />
   515→              </div>
   516→            ) : files.length === 0 ? (
   517→              <div className="flex flex-col items-center justify-center h-64 text-center px-4">
   518→                {searchQuery ? (
   519→                  <><Search className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">No results for &quot;{searchQuery}&quot;</p></>
   520→                ) : activeView === 'trash' ? (
   521→                  <><Trash2 className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">Trash is empty</p></>
   522→                ) : activeView === 'starred' ? (
   523→                  <><Star className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">No starred files</p></>
   524→                ) : (
   525→                  <><FolderOpen className="w-12 h-12 text-slate-300 mb-3" /><p className="text-slate-500">This folder is empty</p>
   526→                  <p className="text-sm text-slate-400 mt-1">Upload files or create new documents</p></>
   527→                )}
   528→              </div>
   529→            ) : viewMode === 'grid' ? (
   530→              <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 xl:grid-cols-6 2xl:grid-cols-8 gap-1 p-4">
   531→                {files.map(file => (
   532→                  <ContextMenu key={file.id}>
   533→                    <ContextMenuTrigger>
   534→                      <div
   535→                        className={`group relative border rounded-lg hover:shadow-md transition-all cursor-pointer ${selectedFiles.includes(file.id) ? 'border-emerald-500 bg-emerald-50/50' : 'border-transparent hover:border-slate-200'}`}
   536→                        onClick={(e) => { e.stopPropagation(); toggleSelect(file.id); setDetailFile(file); }}
   537→                        onDoubleClick={() => handleFileDoubleClick(file)}
   538→                      >
   539→                        <FileThumbnail file={file} />
   540→                        <div className="p-2 flex items-start gap-2">
   541→                          <Checkbox
   542→                            checked={selectedFiles.includes(file.id)}
   543→                            onCheckedChange={() => toggleSelect(file.id)}
   544→                            className="mt-0.5 shrink-0 opacity-0 group-hover:opacity-100 transition-opacity"
   545→                            onClick={(e) => e.stopPropagation()}
   546→                          />
   547→                          <div className="flex-1 min-w-0">
   548→                            <p className="text-sm font-medium truncate">{file.name}</p>
   549→                            <p className="text-xs text-slate-500 mt-0.5">{file.type === 'folder' ? 'Folder' : formatFileSize(file.size)}</p>
   550→                          </div>
   551→                          {file.starred && <Star className="w-3 h-3 text-amber-400 fill-amber-400 shrink-0" />}
   552→                        </div>
   553→                      </div>
   554→                    </ContextMenuTrigger>
   555→                    <ContextMenuContent>
   556→                      <ContextMenuItem onClick={() => handleFileDoubleClick(file)}><Eye className="w-4 h-4 mr-2" /> Open</ContextMenuItem>
   557→                      <ContextMenuItem onClick={() => { setRenameDialog({ open: true, fileId: file.id, name: file.name }); setRenameValue(file.name); }}><Edit3 className="w-4 h-4 mr-2" /> Rename</ContextMenuItem>
   558→                      <ContextMenuItem onClick={() => handleStar(file.id)}>{file.starred ? <><StarOff className="w-4 h-4 mr-2" /> Unstar</> : <><Star className="w-4 h-4 mr-2" /> Star</>}</ContextMenuItem>
   559→                      <ContextMenuItem onClick={() => handleDownloadFile(file)}><Download className="w-4 h-4 mr-2" /> Download</ContextMenuItem>
   560→                      <ContextMenuSeparator />
   561→                      <ContextMenuItem className="text-red-600" onClick={() => {
   562→                        if (activeView === 'trash') handleDeletePermanent([file.id]);
   563→                        else { setSelectedFiles([file.id]); handleDeleteSelected(); }
   564→                      }}><Trash2 className="w-4 h-4 mr-2" /> {activeView === 'trash' ? 'Delete permanently' : 'Move to trash'}</ContextMenuItem>
   565→                    </ContextMenuContent>
   566→                  </ContextMenu>
   567→                ))}
   568→              </div>
   569→            ) : (
   570→              <table className="w-full text-sm">
   571→                <thead className="sticky top-0 bg-white border-b">
   572→                  <tr>
   573→                    <th className="w-10 px-4 py-2"><Checkbox checked={files.length > 0 && selectedFiles.length === files.length} onCheckedChange={(checked) => setSelectedFiles(checked ? files.map(f => f.id) : [])} /></th>
   574→                    <th className="text-left py-2 px-3 font-medium text-slate-500">Name</th>
   575→                    <th className="text-left py-2 px-3 font-medium text-slate-500 hidden md:table-cell">Owner</th>
   576→                    <th className="text-left py-2 px-3 font-medium text-slate-500 hidden sm:table-cell">Modified</th>
   577→                    <th className="text-left py-2 px-3 font-medium text-slate-500 hidden lg:table-cell">Size</th>
   578→                    <th className="w-10 px-2" />
   579→                  </tr>
   580→                </thead>
   581→                <tbody>
   582→                  {files.map(file => (
   583→                    <ContextMenu key={file.id}>
   584→                      <ContextMenuTrigger>
   585→                        <tr
   586→                          className={`border-b hover:bg-slate-50 cursor-pointer transition-colors ${selectedFiles.includes(file.id) ? 'bg-emerald-50' : ''}`}
   587→                          onClick={(e) => { e.stopPropagation(); toggleSelect(file.id); setDetailFile(file); }}
   588→                          onDoubleClick={() => handleFileDoubleClick(file)}
   589→                        >
   590→                          <td className="px-4 py-1"><Checkbox checked={selectedFiles.includes(file.id)} onCheckedChange={() => toggleSelect(file.id)} onClick={(e) => e.stopPropagation()} /></td>
   591→                          <td className="py-1 px-3">
   592→                            <div className="flex items-center gap-3">
   593→                              <FileIcon file={file} />
   594→                              <span className="truncate font-medium">{file.name}</span>
   595→                            </div>
   596→                          </td>
   597→                          <td className="py-1 px-3 text-slate-500 hidden md:table-cell">{user?.name || 'me'}</td>
   598→                          <td className="py-1 px-3 text-slate-500 hidden sm:table-cell">{formatTime(file.updatedAt)}</td>
   599→                          <td className="py-1 px-3 text-slate-500 hidden lg:table-cell">{file.type === 'folder' ? '--' : formatFileSize(file.size)}</td>
   600→                          <td className="px-2">
   601→                            <DropdownMenu>
   602→                              <DropdownMenuTrigger asChild onClick={(e) => e.stopPropagation()}>
   603→                                <Button variant="ghost" size="icon" className="h-8 w-8"><MoreVertical className="w-4 h-4" /></Button>
   604→                              </DropdownMenuTrigger>
   605→                              <DropdownMenuContent align="end">
   606→                                <DropdownMenuItem onClick={() => handleFileDoubleClick(file)}><Eye className="w-4 h-4 mr-2" /> Open</DropdownMenuItem>
   607→                                <DropdownMenuItem onClick={() => { setRenameDialog({ open: true, fileId: file.id, name: file.name }); setRenameValue(file.name); }}><Edit3 className="w-4 h-4 mr-2" /> Rename</DropdownMenuItem>
   608→                                <DropdownMenuItem onClick={() => handleStar(file.id)}>{file.starred ? <><StarOff className="w-4 h-4 mr-2" /> Unstar</> : <><Star className="w-4 h-4 mr-2" /> Star</>}</DropdownMenuItem>
   609→                                <DropdownMenuItem onClick={() => handleDownloadFile(file)}><Download className="w-4 h-4 mr-2" /> Download</DropdownMenuItem>
   610→                                <DropdownMenuSeparator />
   611→                                <DropdownMenuItem className="text-red-600" onClick={() => {
   612→                                  if (activeView === 'trash') handleDeletePermanent([file.id]);
   613→                                  else { setSelectedFiles([file.id]); handleDeleteSelected(); }
   614→                                }}><Trash2 className="w-4 h-4 mr-2" /> {activeView === 'trash' ? 'Delete forever' : 'Move to trash'}</DropdownMenuItem>
   615→                              </DropdownMenuContent>
   616→                            </DropdownMenu>
   617→                          </td>
   618→                        </tr>
   619→                      </ContextMenuTrigger>
   620→                      <ContextMenuContent>
   621→                        <ContextMenuItem onClick={() => handleFileDoubleClick(file)}><Eye className="w-4 h-4 mr-2" /> Open</ContextMenuItem>
   622→                        <ContextMenuItem onClick={() => { setRenameDialog({ open: true, fileId: file.id, name: file.name }); setRenameValue(file.name); }}><Edit3 className="w-4 h-4 mr-2" /> Rename</ContextMenuItem>
   623→                        <ContextMenuItem onClick={() => handleDownloadFile(file)}><Download className="w-4 h-4 mr-2" /> Download</ContextMenuItem>
   624→                        <ContextMenuItem className="text-red-600" onClick={() => {
   625→                          if (activeView === 'trash') handleDeletePermanent([file.id]);
   626→                          else { setSelectedFiles([file.id]); handleDeleteSelected(); }
   627→                        }}><Trash2 className="w-4 h-4 mr-2" /> {activeView === 'trash' ? 'Delete forever' : 'Move to trash'}</ContextMenuItem>
   628→                      </ContextMenuContent>
   629→                    </ContextMenu>
   630→                  ))}
   631→                </tbody>
   632→              </table>
   633→            )}
   634→          </div>
   635→
   636→          {/* Status bar */}
   637→          <div className="flex items-center justify-between px-4 py-1.5 border-t bg-slate-50 text-xs text-slate-500 shrink-0">
   638→            <span>{files.length} item{files.length !== 1 ? 's' : ''}</span>
   639→            <span>Free & unlimited storage</span>
   640→          </div>
   641→        </main>
   642→
   643→        {/* Detail Panel */}
   644→        {detailPanelOpen && detailFile && (
   645→          <aside className="w-72 border-l bg-white p-4 space-y-4 shrink-0 overflow-y-auto">
   646→            <div className="flex items-center justify-between">
   647→              <h3 className="font-semibold text-sm">Details</h3>
   648→              <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setDetailPanelOpen(false)}><X className="w-3.5 h-3.5" /></Button>
   649→            </div>
   650→            <div className="flex flex-col items-center py-4">
   651→              <FileIcon file={detailFile} size={64} />
   652→              <p className="font-medium mt-3 text-center break-all">{detailFile.name}</p>
   653→              <p className="text-xs text-slate-500 mt-1">{detailFile.type === 'folder' ? 'Folder' : getFileCategory(detailFile.mimeType || '', detailFile.name)}</p>
   654→            </div>
   655→            <Separator />
   656→            <div className="space-y-3 text-sm">
   657→              <div><span className="text-slate-500">Size</span><p className="font-medium">{detailFile.type === 'folder' ? '--' : formatFileSize(detailFile.size)}</p></div>
   658→              <div><span className="text-slate-500">Location</span><p className="font-medium break-all">{detailFile.path || '/'}</p></div>
   659→              <div><span className="text-slate-500">Created</span><p className="font-medium">{formatDate(detailFile.createdAt)}</p></div>
   660→              <div><span className="text-slate-500">Modified</span><p className="font-medium">{formatTime(detailFile.updatedAt)}</p></div>
   661→              <div><span className="text-slate-500">Starred</span><p className="font-medium">{detailFile.starred ? 'Yes' : 'No'}</p></div>
   662→              {detailFile.mimeType && <div><span className="text-slate-500">Type</span><p className="font-medium text-xs break-all">{detailFile.mimeType}</p></div>}
   663→            </div>
   664→            <Separator />
   665→            <div className="space-y-2">
   666→              <Button variant="outline" size="sm" className="w-full gap-2" onClick={() => handleFileDoubleClick(detailFile)}><Eye className="w-3.5 h-3.5" /> Open</Button>
   667→              <Button variant="outline" size="sm" className="w-full gap-2" onClick={() => handleDownloadFile(detailFile)}><Download className="w-3.5 h-3.5" /> Download</Button>
   668→              <Button variant="outline" size="sm" className="w-full gap-2" onClick={() => handleStar(detailFile.id)}>{detailFile.starred ? <><StarOff className="w-3.5 h-3.5" /> Unstar</> : <><Star className="w-3.5 h-3.5" /> Star</>}</Button>
   669→            </div>
   670→          </aside>

   672→      </div>
   673→
   674→      {/* New File Dialog */}
   675→      <Dialog open={newFileDialog} onOpenChange={setNewFileDialog}>
   676→        <DialogContent className="sm:max-w-md">
   677→          <DialogHeader>
   678→            <DialogTitle>Create New</DialogTitle>
   679→          </DialogHeader>
   680→          <div className="space-y-4 py-2">
   681→            <div className="space-y-2">
   682→              <label className="text-sm font-medium">File Name</label>
   683→              <Input placeholder="Untitled" value={newFileName} onChange={(e) => setNewFileName(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleCreateFile()} />
   684→            </div>
   685→            <div className="space-y-2">
   686→              <label className="text-sm font-medium">Type</label>
   687→              <div className="grid grid-cols-3 gap-3">
   688→                {[{ type: 'doc' as const, icon: FileText, label: 'Document', desc: 'Rich text editor', color: 'text-blue-600 bg-blue-50' },
   689→                  { type: 'sheet' as const, icon: Table2, label: 'Spreadsheet', desc: 'Cells & formulas', color: 'text-emerald-600 bg-emerald-50' },
   690→                  { type: 'slide' as const, icon: Presentation, label: 'Presentation', desc: 'Slides & themes', color: 'text-orange-600 bg-orange-50' },
   691→                ].map(opt => (
   692→                  <button key={opt.type} onClick={() => setNewFileType(opt.type)} className={`p-3 rounded-lg border-2 text-center transition-all ${newFileType === opt.type ? 'border-emerald-500 bg-emerald-50' : 'border-slate-200 hover:border-slate-300'}`}>
   693→                    <opt.icon className={`w-8 h-8 mx-auto mb-1 ${opt.color.split(' ')[0]}`} />
   694→                    <p className="text-xs font-medium">{opt.label}</p>
   695→                  </button>
   696→                ))}
   697→              </div>
   698→            </div>
   699→          </div>
   700→          <DialogFooter>
   701→            <Button variant="outline" onClick={() => setNewFileDialog(false)}>Cancel</Button>
   702→            <Button onClick={handleCreateFile} disabled={!newFileName.trim()} className="bg-emerald-600 hover:bg-emerald-700">Create</Button>
   703→          </DialogFooter>
   704→        </DialogContent>
   705→      </Dialog>
   706→
   707→      {/* Rename Dialog */}
   708→      <Dialog open={!!renameDialog?.open} onOpenChange={() => setRenameDialog(null)}>
   709→        <DialogContent className="sm:max-w-md">
   710→          <DialogHeader><DialogTitle>Rename</DialogTitle></DialogHeader>
   711→          <Input value={renameValue} onChange={(e) => setRenameValue(e.target.value)} onKeyDown={(e) => e.key === 'Enter' && handleRename()} />
   712→          <DialogFooter>
   713→            <Button variant="outline" onClick={() => setRenameDialog(null)}>Cancel</Button>
   714→            <Button onClick={handleRename} className="bg-emerald-600 hover:bg-emerald-700">Rename</Button>
   715→          </DialogFooter>
   716→        </DialogContent>
   717→      </Dialog>
   718→    </div>
   719→  );
   720→}
